  <meta charset="UTF-8" />

  <title>Telegram Clicker Game</title>

  <meta name="viewport" content="width=device-width, initial-scale=1.0">

  <script src="https://cdn.jsdelivr.net/npm/phaser@3/dist/phaser.min.js"></script>

  <style>

    body { margin: 0; }

    canvas { display: block; margin: 0 auto; }

  </style>

</head>

<body>


<script>

let score = 0;

let scoreText;


const config = {

    type: Phaser.AUTO,

    width: 400,

    height: 300,

    backgroundColor: '#1d1d1d',

    scene: {

        preload: preload,

        create: create,

        update: update

    }

};


function preload() {}


function create() {

    const centerX = this.cameras.main.centerX;

    const centerY = this.cameras.main.centerY;


    scoreText = this.add.text(centerX, centerY - 50, 'Очки: 0', {

        fontSize: '32px',

        color: '#ffffff'

    }).setOrigin(0.5);


    const button = this.add.text(centerX, centerY + 20, 'Нажми меня', {

        fontSize: '28px',

        backgroundColor: '#00aa00',

        color: '#ffffff',

        padding: { x: 20, y: 10 }

    }).setOrigin(0.5).setInteractive();


    button.on('pointerdown', () => {

        score += 10;

        scoreText.setText('Очки: ' + score);

        if (window.TelegramGameProxy) {

            TelegramGameProxy.postEvent("score", score);  // отправка очков

        }

    });

}


function update() {}


const game = new Phaser.Game(config);

</script>
